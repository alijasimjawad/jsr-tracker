function searchSiteAcrossAllScopes(siteId) {
    const q = String(siteId).trim().toLowerCase();
    if (!q) return [];
    const results = [];
    for (const [proj, sections] of Object.entries(db)) {
        if (!(currentUser?.role === 'admin' || hasPerm('view_' + proj))) continue;
        for (const [sec, data] of Object.entries(sections || {})) {
            if (!data || data.length < 2) continue;
            const headers = data[0];
            let siteIdx = headers.findIndex(h => /^site.{0,3}id$/i.test(h));
            if (siteIdx < 0) siteIdx = 0;
            for (let ri = 1; ri < data.length; ri++) {
                const row = data[ri];
                const val = String(row[siteIdx] ?? '').trim().toLowerCase();
                if (val === q) results.push({ proj, sec, headers, row });
            }
        }
    }
    return results;
}

function runSiteLookup() {
    const input = document.getElementById('sl-search-input');
    const q = (input?.value || '').trim();
    const resultsEl = document.getElementById('sl-results');
    const heroEl = document.getElementById('sl-hero');
    if (!q || !resultsEl) return;

    const matches = searchSiteAcrossAllScopes(q);
    if (heroEl) heroEl.style.display = 'none';

    if (matches.length === 0) {
        resultsEl.innerHTML = `<div class="al-empty">No site found matching "${esc(q)}" in any project scope.</div>`;
        return;
    }

    resultsEl.innerHTML = matches.map(m => {
        const secName = sectionLabels[m.proj]?.[m.sec] || customSections[m.proj]?.[m.sec] || LABELS[m.sec] || m.sec;
        const fields = m.headers.map((h, i) => {
            const val = m.row[i];
            if (/^site.{0,3}id$/i.test(h) || String(val ?? '').trim() === '') return '';
            return `<div class="sl-field"><span class="sl-field-k">${esc(String(h))}</span><span class="sl-field-v">${esc(String(val))}</span></div>`;
        }).join('');
        return `
            <div class="sl-card">
                <div class="sl-card-head">
                    <span class="sl-card-proj">${esc(PROJ_NAMES[m.proj] || m.proj)}</span>
                    <span class="sl-card-sec">${esc(String(secName))}</span>
                    <span class="sl-card-site">Site ${esc(q)}</span>
                </div>
                <div class="sl-card-body">${fields}</div>
            </div>`;
    }).join('');
}
